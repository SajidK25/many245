from flask import Flask, render_template, request
from models import db, RelayData

app = Flask(__name__)

@app.route('/relay_data', methods=['GET'])
def relay_data():
    # Collect filter parameters from the form
    search_filters = {
        'id': request.args.get('id'),
        'location': request.args.get('location'),
        'driver': request.args.get('driver'),
        'domicile': request.args.get('domicile'),
        'disruption': request.args.get('disruption'),
        'work_type': request.args.get('work_type'),
        'program': request.args.get('program')
    }

    # Start building the query
    query = RelayData.query

    # Apply filters
    for field, value in search_filters.items():
        if value:
            query = query.filter(getattr(RelayData, field).ilike(f"%{value}%"))

    filtered_data = query.all()

    return render_template("relay_data.html", relay_data=filtered_data)

if __name__ == '__main__':
    app.run(debug=True)

How It Works:
User fills the search form (by IDs, location, drivers, domiciles, disruptions, work types, or programs).
Flask processes the filters using the GET method.
Filtered data is displayed in the table.
Partial matches are supported using ilike for case-insensitive search.

Real-Time Filtering with JavaScript (Optional)
If you want real-time filtering without page reloads, add this inside the <head>:



<script>
    function filterTable() {
        let input = document.getElementById("searchInput").value.toUpperCase();
        let table = document.getElementById("dataTable");
        let tr = table.getElementsByTagName("tr");

        for (let i = 1; i < tr.length; i++) {
            let show = false;
            let td = tr[i].getElementsByTagName("td");
            for (let j = 0; j < td.length; j++) {
                if (td[j] && td[j].innerHTML.toUpperCase().includes(input)) {
                    show = true;
                    break;
                }
            }
            tr[i].style.display = show ? "" : "none";
        }
    }
</script>


Add this search box above the table:

<input type="text" id="searchInput" onkeyup="filterTable()" placeholder="Real-time search...">
